cmake_minimum_required(VERSION 3.16)
project(imu_driver_host_tests LANGUAGES C CXX)
enable_testing()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(URU_COMPONENTS_DIR "" CACHE PATH "URU components directory containing rtos and vectors")
if(NOT EXISTS "${URU_COMPONENTS_DIR}/rtos/include/rtos/Spi.hpp")
    message(FATAL_ERROR "Set URU_COMPONENTS_DIR to initialized URU firmware components")
endif()
add_executable(fifo_temperature_test fifo_temperature_test.cpp ../lsm6dsv80x_reg.c)
target_include_directories(fifo_temperature_test PRIVATE
    .. ../include
    "${URU_COMPONENTS_DIR}/rtos/include"
    "${URU_COMPONENTS_DIR}/vectors/include")
# Unused platform entry points are outside this register-level host test.
target_compile_options(fifo_temperature_test PRIVATE -ffunction-sections -fdata-sections)
target_link_options(fifo_temperature_test PRIVATE -Wl,--gc-sections)
add_test(NAME fifo_temperature COMMAND fifo_temperature_test)
foreach(part IN ITEMS 80 32)
    foreach(test IN ITEMS contract rotation)
        add_executable(${test}_${part} ${test}_test.cpp ../lsm6dsv80x.cpp ../lsm6dsv${part}x_reg.c)
        target_include_directories(${test}_${part} PRIVATE .. ../include "${URU_COMPONENTS_DIR}/rtos/include" "${URU_COMPONENTS_DIR}/vectors/include")
        target_compile_definitions(${test}_${part} PRIVATE IMU_LSM6DSV_PART=${part})
        add_test(NAME ${test}_${part} COMMAND ${test}_${part})
    endforeach()
endforeach()
add_executable(baseline_trace baseline_trace.cpp baseline/lsm6dsv80x.cpp ../lsm6dsv80x_reg.c)
target_include_directories(baseline_trace PRIVATE baseline .. "${URU_COMPONENTS_DIR}/rtos/include" "${URU_COMPONENTS_DIR}/vectors/include")
add_executable(current_trace baseline_trace.cpp ../lsm6dsv80x.cpp ../lsm6dsv80x_reg.c)
target_include_directories(current_trace PRIVATE ../include .. "${URU_COMPONENTS_DIR}/rtos/include" "${URU_COMPONENTS_DIR}/vectors/include")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME healthy_80x_parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/compare_trace.py $<TARGET_FILE:baseline_trace> $<TARGET_FILE:current_trace>)
foreach(target IN ITEMS contract_80 contract_32 rotation_80 rotation_32 current_trace)
    target_compile_options(${target} PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror>)
endforeach()
foreach(part IN ITEMS 80 32)
    add_test(NAME selection_${part} COMMAND ${CMAKE_COMMAND} -DIMU_LSM6DSV_PART=${part} -P ${CMAKE_CURRENT_SOURCE_DIR}/selection_test.cmake)
endforeach()
add_test(NAME selection_invalid COMMAND ${CMAKE_COMMAND} -DIMU_LSM6DSV_PART=16 -P ${CMAKE_CURRENT_SOURCE_DIR}/selection_test.cmake)
set_tests_properties(selection_invalid PROPERTIES WILL_FAIL TRUE)
add_test(NAME selected_symbols COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/selected_symbols.py $<TARGET_FILE:contract_80> $<TARGET_FILE:contract_32>)
add_executable(neutral_trace baseline_trace.cpp ../lsm6dsv80x.cpp ../lsm6dsv80x_reg.c)
target_compile_definitions(neutral_trace PRIVATE NEUTRAL_TRACE=1)
target_include_directories(neutral_trace PRIVATE ../include .. "${URU_COMPONENTS_DIR}/rtos/include" "${URU_COMPONENTS_DIR}/vectors/include")
target_compile_options(neutral_trace PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Werror>)
add_test(NAME neutral_80x_parity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/compare_trace.py $<TARGET_FILE:baseline_trace> $<TARGET_FILE:neutral_trace>)
# Compile consumer TUs independently: do not inherit the driver's part define.
foreach(part IN ITEMS 80 32)
    add_library(link_driver_${part} STATIC ../lsm6dsv80x.cpp ../lsm6dsv${part}x_reg.c)
    target_compile_definitions(link_driver_${part} PRIVATE IMU_LSM6DSV_PART=${part})
    target_include_directories(link_driver_${part} PRIVATE .. ../include "${URU_COMPONENTS_DIR}/rtos/include" "${URU_COMPONENTS_DIR}/vectors/include")
    add_test(NAME part_link_${part} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/part_link_test.py
        ${CMAKE_CXX_COMPILER} ${part} $<TARGET_FILE:link_driver_${part}> ${URU_COMPONENTS_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../include "${CMAKE_CXX_FLAGS}" "${CMAKE_EXE_LINKER_FLAGS}")
endforeach()
